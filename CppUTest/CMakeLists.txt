# CppUTest
set(CPPUTEST_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CPPUTEST_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CPPUTEST_EXTENSIONS OFF CACHE BOOL "" FORCE)

set(CPPUTEST_MEM_LEAK_DETECTION_DISABLED ON CACHE BOOL "" FORCE)
set(CPPUTEST_STD_CPP_LIB_DISABLED ON CACHE BOOL "" FORCE)
set(CPPUTEST_USE_LONG_LONG OFF CACHE BOOL "" FORCE)

# Bare-metal target: these facilities are not available.
set(CPPUTEST_HAVE_KILL OFF CACHE INTERNAL "" FORCE)
set(CPPUTEST_HAVE_FORK OFF CACHE INTERNAL "" FORCE)
set(CPPUTEST_HAVE_WAITPID OFF CACHE INTERNAL "" FORCE)
set(CPPUTEST_HAVE_GETTIMEOFDAY OFF CACHE INTERNAL "" FORCE)
set(CPPUTEST_HAVE_PTHREAD_MUTEX_LOCK OFF CACHE INTERNAL "" FORCE)

# Do not let CppUTest's FindThreads add pthreads.
set(CMAKE_DISABLE_FIND_PACKAGE_Threads TRUE)

include(FetchContent)
FetchContent_Declare(
    CppUTest
    GIT_REPOSITORY https://github.com/cpputest/cpputest.git
    GIT_TAG        82c7db525f2da2d50944463758d3183b0ede2338
)

FetchContent_MakeAvailable(CppUTest)
